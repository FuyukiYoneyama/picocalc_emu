# Canonical PicoCalc BSP — host build.
#
# Builds the BSP's public API against host models instead of the RP2040,
# so application logic can be tested natively. This is a *separate*
# library from `picocalc_bsp`: the device build is hardware-verified and
# nothing here is allowed to change it.
#
# Note which sources are shared rather than reimplemented. The filesystem
# layer and its FatFs bridge have no Pico SDK dependency at all -- they
# speak only to `picocalc::sdcard` and to ChanFatFS -- so the host build
# compiles the very same files the device does, on top of a RAM block
# device. A host filesystem test therefore exercises the shipping code,
# not a stand-in for it.

cmake_minimum_required(VERSION 3.13)

set(PICOCALC_BSP_ROOT ${CMAKE_CURRENT_LIST_DIR}/..)

file(READ "${PICOCALC_BSP_ROOT}/VERSION" PICOCALC_BSP_VERSION)
string(STRIP "${PICOCALC_BSP_VERSION}" PICOCALC_BSP_VERSION)
if(NOT DEFINED PICOCALC_APP_VERSION)
    set(PICOCALC_APP_VERSION "host")
endif()

add_library(picocalc_bsp_host STATIC
    # Host models: everything that touches a pin on the device.
    src/audio.cpp
    src/bsp.cpp
    src/display.cpp
    src/framebuffer_out.cpp
    src/keyboard.cpp
    src/psram.cpp
    src/runtime.cpp
    src/sdcard.cpp
    # Shared with the device build, unmodified.
    ${PICOCALC_BSP_ROOT}/src/fatfs_diskio.cpp
    ${PICOCALC_BSP_ROOT}/src/filesystem.cpp
    ${PICOCALC_BSP_ROOT}/third_party/ChanFatFS/src/ff.c
    ${PICOCALC_BSP_ROOT}/third_party/ChanFatFS/src/ffunicode.c
)

target_include_directories(picocalc_bsp_host PUBLIC
    # The host SDK shim comes first so `pico/stdlib.h` resolves here.
    ${CMAKE_CURRENT_LIST_DIR}/include
    ${PICOCALC_BSP_ROOT}/include
    ${PICOCALC_BSP_ROOT}/third_party/ChanFatFS/include
)

target_compile_features(picocalc_bsp_host PUBLIC cxx_std_17)

target_compile_definitions(picocalc_bsp_host PUBLIC
    # Matches the device build. Without it fatfs_diskio.cpp answers
    # RES_WRPRT and the card is silently read-only -- which is how the
    # first host run of the BSP's own smoke test failed, at the write
    # stage, with no sectors ever reaching the block layer.
    PICOCALC_BSP_ENABLE_SD_WRITE=1
    PICOCALC_BSP_VERSION="${PICOCALC_BSP_VERSION}"
    PICOCALC_APP_VERSION="${PICOCALC_APP_VERSION}"
    # Application code can ask which backend it is being built for. The
    # answer should almost never change what an app does -- if it does,
    # the host result stops standing for the device one.
    PICOCALC_HOST_BUILD=1
)

# Same local suppression the device build uses: GCC 13 reports a
# false positive in ChanFatFS's long-file-name collision helper. Kept to
# that one source rather than weakening warnings for BSP or app code.
set_source_files_properties(
    ${PICOCALC_BSP_ROOT}/third_party/ChanFatFS/src/ff.c
    PROPERTIES COMPILE_OPTIONS "-Wno-stringop-overflow"
)

target_compile_options(picocalc_bsp_host PRIVATE -Wall -Wextra)
